1.Create a table called Restaurant with columns: id, name, location, and cuisine. Apply NOT NULL constraint to name and location, and make id the PRIMARY KEY.

CREATE TABLE Restaurant (
    id       INT PRIMARY KEY AUTO_INCREMENT,
    name     VARCHAR(100) NOT NULL,
    location VARCHAR(100) NOT NULL,
    cuisine  VARCHAR(50)
);

2.Create a table called FoodOrder with columns: order_id, restaurant_id, user_id, and order_total. Set order_id as PRIMARY KEY, and add a FOREIGN KEY constraint on restaurant_id referencing Restaurant(id).

CREATE TABLE FoodOrder (
    order_id      INT PRIMARY KEY AUTO_INCREMENT,
    restaurant_id INT,
    user_id       INT,
    order_total   DECIMAL(10,2),
    FOREIGN KEY (restaurant_id) REFERENCES Restaurant(id)
);

CREATE TABLE SpotifyUser (
    user_id           INT PRIMARY KEY AUTO_INCREMENT,
    username          VARCHAR(50)  UNIQUE,

4.Create a VIEW named TopSpendersView that shows usernames and order_total from a FoodOrder table where order_total is greater than 1000.

CREATE VIEW TopSpendersView AS
SELECT u.username,
       f.order_total
FROM FoodOrder f
JOIN SpotifyUser u ON f.user_id = u.user_id
WHERE f.order_total > 1000;

5.Write a SQL query to list all users from the TopSpendersView who have spent more than 2000, sorted by order_total descending.<br><br><em><strong>Constraint:</strong> Only use the TopSpendersView for your query, not the original FoodOrder table.</em>

SELECT username, order_total
FROM TopSpendersView
WHERE order_total > 2000
ORDER BY order_total DESC;
